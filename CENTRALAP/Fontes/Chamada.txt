//------------------------------------------------------------
// Chama Formulario de Controle de Pendencias
Function PedeSenha(wIdUsuario:Integer;wIdAuto:Integer;Var wSenhaUsada,wObsSenha:String):Boolean;
Begin
// Cria Formulario
  Application.CreateForm(TFrmSenha,FrmSenha);
  FrmSenha.wIdUsuario :=wIdUsuario;
  FrmSenha.wIdAuto    :=wIdAuto;
  FrmSenha.wTipo      :='A';
  FrmSenha.wResult    :=False;
  FrmSenha.ShowModal;
// Senha Nao Confirmada
  If FrmSenha.wResult = False Then Begin
    wSenhaUsada :='';
    wObsSenha   :='';
    Result      :=False;
    FrmSenha.Free;
    Exit;
  End;
  wSenhaUsada:=FrmSenha.wSenhaUsada;
  wObsSenha  :=FrmSenha.Memo1.Text;
  Result     :=True;
  FrmSenha.Free;
End;
