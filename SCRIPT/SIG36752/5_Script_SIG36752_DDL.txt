-- TABELA
CREATE TABLE CM.PERCFAIXA (
  IDPERCFAIXA       NUMBER  NOT NULL,
  IDFAIXA           NUMBER  NOT NULL,
  DATAINIVIG        DATE    NOT NULL,
  FAIXA             NUMBER  NOT NULL,
  PERCPARTAT        NUMBER  NOT NULL,
  PERCPATROAT       NUMBER  NOT NULL,
  PERCPARTAS        NUMBER  NOT NULL,
  PERCPATROAS       NUMBER  NOT NULL,  
  TRGUSERINCLUSAO   VARCHAR2(30) DEFAULT USER,
  TRGDTINCLUSAO     DATE         DEFAULT SYSDATE
);

-- PK
ALTER TABLE CM.PERCFAIXA ADD CONSTRAINT PK_PERCFAIXA 
   PRIMARY KEY(IDPERCFAIXA);

-- FK
ALTER TABLE CM.PERCFAIXA ADD CONSTRAINT FK_FAIXA FOREIGN KEY(IDFAIXA) 
  REFERENCES CM.FAIXA (IDFAIXA);

-- criar sequence
CREATE SEQUENCE CM.SEQPERCFAIXA
  INCREMENT BY 1
  START WITH 1
  MINVALUE 1
  MAXVALUE 999999999999999999999999999
  NOCYCLE
  NOORDER
  NOCACHE;

-- SINONIMO
CREATE OR REPLACE PUBLIC SYNONYM PERCFAIXA    FOR CM.PERCFAIXA;
CREATE OR REPLACE PUBLIC SYNONYM SEQPERCFAIXA FOR CM.SEQPERCFAIXA;

-- COMENTARIOS
COMMENT ON TABLE CM.PERCFAIXA  IS 'Cadastro de Percentuais da Faixa';

COMMENT ON COLUMN CM.PERCFAIXA.IDPERCFAIXA        IS 'SEQUENCIAL IDENTIFICADOR INTERNO DO PERCENTUAL DA FAIXA';
COMMENT ON COLUMN CM.PERCFAIXA.IDFAIXA            IS 'SEQUENCIAL IDENTIFICADOR INTERNO DA FAIXA';
COMMENT ON COLUMN CM.PERCFAIXA.DATAINIVIG         IS 'DATA INICIO DA VIGENCIA DA FAIXA';
COMMENT ON COLUMN CM.PERCFAIXA.FAIXA              IS 'NÚMERO DA FAIXA';
COMMENT ON COLUMN CM.PERCFAIXA.PERCPARTAT         IS 'PERCENTUAL DO PARTICIPANTE ATIVO';
COMMENT ON COLUMN CM.PERCFAIXA.PERCPATROAT        IS 'PERCENTUAL DA PATROCINADORA ATIVO';
COMMENT ON COLUMN CM.PERCFAIXA.PERCPARTAS         IS 'PERCENTUAL DO PARTICIPANTE ASSISTIDO';
COMMENT ON COLUMN CM.PERCFAIXA.PERCPATROAS        IS 'PERCENTUAL DA PATROCINADORA ASSISTIDO';
COMMENT ON COLUMN CM.PERCFAIXA.TRGUSERINCLUSAO    IS 'TRIGUER DO USUÁRIO DE INCLUSÃO.';
COMMENT ON COLUMN CM.PERCFAIXA.TRGDTINCLUSAO      IS 'TRIGUER DA DATA DE INCLUSÃO.';
