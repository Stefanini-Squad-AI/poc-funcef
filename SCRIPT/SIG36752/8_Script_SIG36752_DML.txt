-- Declaração das Variaveis


declare

  vIdForm      Integer;
  vIdFuncao    Integer;
  vIdObjeto    Integer;
  vIdOperFunc  Integer;

  vIdModulo    Integer;
  vIdFormPai   Integer;
  vIdFuncaoPai Integer;
  vIdOperacao  Integer;

  vIdModulo    := 487;
  vIdFormPai   := 13559;
  vIdFuncaoPai := 27073;
  
  -- busca/inclui form para autorizar componentes
  begin
    -- Select
    Select IdForm Into vIdFormPai from Form
    where NomeForm = 'FrmCadFaixasPercentuais' 
      and idmodulo = vIdModulo;
    -- Exception
    EXCEPTION WHEN OTHERS THEN vIdFormPai := -1;
    If vIdFormPai = -1 then
       Select Max(idform)+1 Into vIdFormPai From form;
       insert into form (idform, nomeform, idmodulo, descform) values (vIdFormPai, 'FrmCadFaixasPercentuais', vIdModulo, 'Faixa e Percentuais');
    End If;
  end;


  -- Validação da Variavel Sequencial da Tabela Funcao
  Begin
    -- Select
    Select Idfuncao Into vIdFuncaoPai From Funcao
    where NomeFuncao = 'Faixas e Percentuais'
      and IdModulo = vIdModulo;
    -- Exception
    EXCEPTION WHEN OTHERS THEN vIdFuncao := -1;
    -- Criação do Registro
    If vIdFuncao = -1 then
      Select Max(IdFuncao)+1 Into vIdFuncao From Funcao;
      Insert Into Funcao(IdFuncao,NomeFuncao,IdModulo,IdFuncaoPai)
                  values(vIdFuncao,'Faixas e Percentuais', vIdModulo, 25874);
    End If;
  End;

  -- INCLUSAO
  -------------------------------------------------------------------------

    End If;
  End;  
  
  -- INSERE OPERACAO: INCLUSÃO
  Begin
    -- Select
    Select idoperacao Into vIdOperacao from operacao
    where TRIM(nomeoperacao) = 'HABILITAR INCLUSAO';
    -- Exception
    EXCEPTION WHEN OTHERS THEN vIdOperacao := -1;
    -- Criação do Registro
    If vIdOperacao = -1 then
      Select Max(idoperacao)+1 Into vIdOperacao From operacao;
      Insert Into Operacao(IdOperacao,NomeOperacao) Values( vIdOperacao, 'HABILITAR INCLUSAO');
    End If;
  End;

  -- Validação da Variavel Sequencial da Tabela OperFunc
  Begin
    -- Select
    Select IdOperFunc into vIdOperFunc from OperFunc
    where idModulo = vIdModulo
      and idOperacao = vIdOperacao
      and IdFuncao = vIdFuncaoPai;
    -- Exception
    EXCEPTION WHEN OTHERS THEN vIdOperFunc := -1;
    -- Criação do Registro
    If vIdOperFunc = -1 then
      Select Max(IdOperFunc)+1 Into vIdOperFunc from OperFunc;
      Insert Into Operfunc(idOperfunc,Idmodulo,Idoperacao,Idfuncao)
                    Values(vIdOperFunc, vIdModulo, vIdOperacao, vIdFuncaoPai);
      Insert into frobfnop(idOperfunc,idobjeto,idform)
                    values(vIdOperFunc, vIdObjeto, vIdFormPai);
    End If;
  End;

  -- ALTERACAO
  -------------------------------------------------------------------------

                    values(vIdOperFunc, vIdObjeto, vIdFormPai);
    End If;
  End;

    -- Exception

      Select Max(IdOperFunc)+1 Into vIdOperFunc from OperFunc;

  Begin
    -- Select
    Select IdObjeto Into vIdObjeto from Objeto

    -- Exception
    EXCEPTION WHEN OTHERS THEN vIdObjeto := -1;

  Begin
    -- Select
    Select IdOperFunc into vIdOperFunc from OperFunc
    where idModulo = vIdModulo

      Insert Into Operfunc(idOperfunc,Idmodulo,Idoperacao,Idfuncao)
                    Values(vIdOperFunc, vIdModulo, vIdOperacao, vIdFuncaoPai);
      Insert into frobfnop(idOperfunc,idobjeto,idform)
                    values(vIdOperFunc, vIdObjeto, vIdFormPai);
    End If;
  End;
  -- Grava Todas Informações no Banco
  Commit;
end;
