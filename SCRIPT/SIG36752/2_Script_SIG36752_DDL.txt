declare 
  -- Local variables here
  i integer;
  v_maiorid    number := 0;
begin
  -- Test statements here

  Select max(idtpcontribuicao)+1 into v_maiorid
    from TPCONTRIBUICAO;

  if (v_maiorid > 0) then   
    
     
      SELECT COUNT(*) INTO i
      FROM all_sequences
      WHERE sequence_name = 'SEQTPCONTRIBUICAO'
      AND sequence_owner = 'CM';
      IF i > 0 THEN
         execute immediate 'drop sequence CM.SEQTPCONTRIBUICAO'; 
      END IF;

       execute immediate 'create sequence CM.SEQTPCONTRIBUICAO 
                          minvalue 1
                          maxvalue 999999999999999999999999999
                          start with '||v_maiorid||'
                          increment by 1
                          nocache';

       execute immediate 'create or replace public synonym SEQTPCONTRIBUICAO FOR CM.SEQTPCONTRIBUICAO';
      
  end if;

end;
