-- TABELA
CREATE TABLE CM.FAIXA (
  IDFAIXA           NUMBER  NOT NULL,
  DESCRICAO         VARCHAR2(60) NOT NULL,
  IDPESSJUR         NUMBER  NOT NULL,
  IDPLANOPREV       NUMBER  NOT NULL,
  IDPLANPREVCONTAB  NUMBER  NOT NULL,
  IDTPCONTRIBUICAO  NUMBER  NOT NULL,
  TRGUSERINCLUSAO   VARCHAR2(30) DEFAULT USER,
  TRGDTINCLUSAO     DATE         DEFAULT SYSDATE
);

-- PK
ALTER TABLE CM.FAIXA ADD CONSTRAINT PK_FAIXA 
   PRIMARY KEY(IDFAIXA);

-- criar sequence
CREATE SEQUENCE CM.SEQFAIXA
  INCREMENT BY 1
  START WITH 1
  MINVALUE 1
  MAXVALUE 999999999999999999999999999
  NOCYCLE
  NOORDER
  NOCACHE;

-- SINONIMO
CREATE OR REPLACE PUBLIC SYNONYM FAIXA FOR CM.FAIXA;
CREATE OR REPLACE PUBLIC SYNONYM SEQFAIXA FOR CM.SEQFAIXA;

-- COMENTARIOS
COMMENT ON TABLE CM.FAIXA  IS 'Cadastro de Faixas';

COMMENT ON COLUMN CM.FAIXA.IDFAIXA            IS 'SEQUENCIAL IDENTIFICADOR INTERNO DA FAIXA.';
COMMENT ON COLUMN CM.FAIXA.DESCRICAO          IS 'DESCRICAO DA FAIXA.';
COMMENT ON COLUMN CM.FAIXA.IDPESSJUR          IS 'IDENTIFICADOR DA PATROCINADORA.';
COMMENT ON COLUMN CM.FAIXA.IDPLANOPREV        IS 'IDENTIFICADOR DO PLANO PREVIDENCIÁRIO.';
COMMENT ON COLUMN CM.FAIXA.IDPLANPREVCONTAB   IS 'IDENTIFICADOR DO PLANO PREVIDENCIÁRIO.';
COMMENT ON COLUMN CM.FAIXA.IDTPCONTRIBUICAO   IS 'IDENTIFICADOR DO TIPO DE CONTRIBUIÇÃO.';
COMMENT ON COLUMN CM.FAIXA.TRGUSERINCLUSAO    IS 'TRIGUER DO USUÁRIO DE INCLUSÃO.';
COMMENT ON COLUMN CM.FAIXA.TRGDTINCLUSAO      IS 'TRIGUER DA DATA DE INCLUSÃO.';
