select u.nomeusuario, u.descricao, g.nomegrupo, d.name, t.table_name, c.Column_Name
from usuariosistema u, grupoacesso g, grupousu x, dataviewacesso v, dataview d, tabelaacesso t, colunaacesso c
where u.idusuario = 4
  and u.idusuario = x.idusuario
  and x.idgrupo = g.idgrupo
  and u.idespacesso = v.idespacesso(+)
  and v.iddataview  = d.iddataview(+) 
  and v.origemcmdv  = d.origemcmdv(+)
  and u.idespacesso = t.idespacesso(+)
  and t.idespacesso = c.idespacesso(+)
  and t.table_name  = c.table_name(+);
