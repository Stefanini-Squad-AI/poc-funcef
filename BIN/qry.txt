SELECT DISTINCT
  P.IDPESSOA AS IDPESSOA,
  P.NOME AS FUNCIONARIO,
  P.NUMDOCUMENTO AS CPF,
  CR.DESCRICAO AS CURSO,
  CR.IDCURSO, 
(H.DUR_PRAT + H.DUR_TEOR) AS DURACAO,
  '0' || RTRIM(UPPER(P.NOME)) AS IDENT,
  F.MATRICULA,
  '10° CONGRESSO INTERNACIONAL DE SHOPPING CENTERS' AS DESCRICAO,
  'HWCJZRQPQQ YIGQXGDEVN' AS ENTIDADE,
  '' AS INSTRUTOR,
  '30/7/2012' AS DATAINI,
  '3/8/2012' AS DATAFIM,
  RTRIM(PJ.NOME) || ' - ' || RTRIM(CC.NOME) AS ENDSETOR,
  C.TITULO AS CARGO,
  H.LOCALCURSO AS LOCAL,
  H.DATAHORA,
  H.INSTRUTORES
FROM
  PESSOA P, PESSOA PJ, HSTTRN H,
  FUNCIONARIO F, CENTCUST CC, CARGO C, CURSO CR 
WHERE
  (H.IDCURSO         = 1525) AND
  (H.IDENTIDINSTR    = 304428) AND
  (H.DATPLINI        = TO_DATE('30/7/2012','DD/MM/YYYY')) AND
  (H.DATPLFIM        = TO_DATE('3/8/2012','DD/MM/YYYY')) AND
  (H.DATREINI        = TO_DATE('30/7/2012','DD/MM/YYYY')) AND
  (H.DATREFIM        = TO_DATE('3/8/2012','DD/MM/YYYY')) AND
  (H.IDPESSOA       IN (1024496)) AND
  (H.IDPESSOA        = P.IDPESSOA) AND
  (H.IDPESSOA        = F.IDPESSOA) AND
  (F.IDESTAB         = PJ.IDPESSOA) AND
  (F.IDEMPRESA       = CC.IDEMPRESA) AND
  (F.CODCENTROCUSTO  = CC.CODCENTROCUSTO) AND
  (F.IDCARGO         = C.IDCARGO) AND 
  (CR.IDCURSO        = H.IDCURSO)
UNION
(
SELECT
  P.IDPESSOA AS IDPESSOA,
  P.NOME AS FUNCIONARIO,
  P.NUMDOCUMENTO AS CPF,
  CR.DESCRICAO AS CURSO,
  CR.IDCURSO, 
(H.DUR_PRAT + H.DUR_TEOR) AS DURACAO,
  '1' || RTRIM(UPPER(P.NOME)) AS IDENT,
  TO_CHAR(F.IDPESSOA) AS MATRICULA,
  '10° CONGRESSO INTERNACIONAL DE SHOPPING CENTERS' AS DESCRICAO,
  'HWCJZRQPQQ YIGQXGDEVN' AS ENTIDADE,
  '' AS INSTRUTOR,
  '30/7/2012' AS DATAINI,
  '3/8/2012' AS DATAFIM,
  RTRIM(EP.LOGRADOURO) || ' ' || TO_CHAR(EP.NUMERO) || ' ' ||
  RTRIM(EP.COMPLEMENTO) || ' ' || RTRIM(EP.BAIRRO) || ' ' || RTRIM(EP.CEP) ||
  ' ' || RTRIM(CI.NOME) || ' ' || RTRIM(EP.CODESTADO) || ' - Tel: ' ||
  RTRIM(TELEFONE.DDI) || TELEFONE.DDD || ' ' || TELEFONE.NUMERO AS ENDSETOR,
  C.TITULO AS CARGO,
  H.LOCALCURSO AS LOCAL,
  H.DATAHORA,
  H.INSTRUTORES
FROM
  PESSOA P, HSTTRN H, CANDIDAT F, ENDPESS EP, CIDADES CI, CARGO C, CURSO CR, 
  (SELECT
     TE.IDENDERECO, TE.IDTELEFONE, TE.DDD, TE.DDI, TE.NUMERO
   FROM
     TELENDPESS TE,
     (SELECT
        MIN(IDTELEFONE) AS IDTELEFONE, IDENDERECO
      FROM
        TELENDPESS
      GROUP BY
        IDENDERECO) ENDER
   WHERE
     (ENDER.IDTELEFONE = TE.IDTELEFONE)) TELEFONE
WHERE
  (H.IDCURSO          = 1525) AND
  (H.IDENTIDINSTR     = 304428) AND
  (H.DATPLINI         = TO_DATE('30/7/2012','DD/MM/YYYY')) AND
  (H.DATPLFIM         = TO_DATE('3/8/2012','DD/MM/YYYY')) AND
  (H.DATREINI         = TO_DATE('30/7/2012','DD/MM/YYYY')) AND
  (H.DATREFIM         = TO_DATE('3/8/2012','DD/MM/YYYY')) AND
  (H.IDPESSOA        IN (1024496)) AND
  (H.IDPESSOA         = P.IDPESSOA) AND
  (H.IDPESSOA         = F.IDPESSOA) AND
  (P.IDPESSOA         = EP.IDPESSOA(+)) AND
  (P.IDENDRESIDENCIAL = EP.IDENDERECO(+)) AND
  (EP.IDCIDADES       = CI.IDCIDADES(+)) AND
  (P.IDENDRESIDENCIAL = TELEFONE.IDENDERECO(+)) AND
  (F.IDCARGO          = C.IDCARGO) AND 
  (CR.IDCURSO         = H.IDCURSO) 
)
ORDER BY 2
