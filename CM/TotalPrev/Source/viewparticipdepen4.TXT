
CREATE OR REPLACE VIEW "CM"."VWPARTICIPDEPEN" AS
SELECT
  E.MATRICULA,
  P.IDPESSOA AS IDTITULAR,
  P.IDPESSOA AS IDPESSOA,
  P.NOME,
  'PRP' AS IDDEPENDENCIA,
  P.NUMDOCUMENTO,
  E.IDPESSJUR,
  PP.INSCRICAONUMERO,
  PP.IDPLANOPREV,
  PP.IDSITPART,
  PP.SEQPROPOSTA,
  PP.FLGDESATIVADO,
  PPREV.NOME AS PLANO,
  PPREV.IDRGELEGBENEF,
  PJ.NOME AS PATRO,
  SIT.DESCRICAO,
  SF.DESCRICAO AS SITFUND,
  P.EMAIL
FROM PARTPREVPLAN PP, ELEGPATRO E, PESSOA P, PLANPREV PPREV, PESSOA PJ, SITPART SIT, SITFUNC SF
WHERE   E.IDPESSOA = P.IDPESSOA  AND

        E.IDPESSOA = E.IDPESSOA  AND
        P.IDPESSOA = P.IDPESSOA  AND
        PP.IDPESSOA = PP.IDPESSOA  AND
        PP.INSCRICAONUMERO = PP.INSCRICAONUMERO AND 

        PJ.IDPESSOA = PJ.IDPESSOA  AND
        SF.IDSITFUNC(+)   = E.IDSITFUNC AND

        PP.IDPESSOA(+)    = E.IDPESSOA   AND
        PP.IDPESSJUR(+)   = E.IDPESSJUR   AND
        (PP.IDPESSOA      = P.IDPESSOA OR PP.IDPESSOA IS NULL)  AND
        E.IDPESSOA        = P.IDPESSOA AND
        PP.IDPLANOPREV    = PPREV.IDPLANOPREV(+) AND
        E.IDPESSJUR       = PJ.IDPESSOA(+) AND
        PP.IDSITPART      = SIT.IDSITPART(+)

UNION

SELECT
  NVL(DP.MATRICULA, EL.MATRICULA) AS MATRICULA,
  DP.IDTITULAR AS IDTITULAR,
  DP.IDPESSOA AS IDPESSOA,
  P.NOME,
  IDDEPENDENCIA,
  P.NUMDOCUMENTO,
  0 AS IDPESSJUR,
  0 AS INSCRICAONUMERO,
  0 AS IDPLANOPREV,
  0 AS IDSITPART,
  9 AS FLGDESATIVADO,
  0 AS SEQPROPOSTA,
  '' AS PLANO,
  0 AS IDRGELEGBENEF,
  '' AS PATRO,
  '' AS DESCRICAO,
  '' AS SITFUND,
  P.EMAIL
FROM DEPENTIT DP, PESSOA P, ELEGPATRO EL
WHERE 
      EL.IDPESSOA = DP.IDTITULAR  AND 
      EL.IDPESSOA = EL.IDPESSOA   AND
      DP.IDDEPENDENCIA <> 'PRP'   AND
      DP.IDPESSOA  = P.IDPESSOA   AND
      DP.IDPESSOA <> DP.IDTITULAR AND
      DP.IDPESSOA  = DP.IDPESSOA  AND
      DP.IDTITULAR = DP.IDTITULAR AND
      P.IDPESSOA   = P.IDPESSOA
